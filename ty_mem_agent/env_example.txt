# === AutoGLM 手机操作配置（用于订票、购物、外卖、打车等手机APP操作）===
# 项目地址: https://github.com/zai-org/Open-AutoGLM
# 
# AutoGLM 是手机Agent框架，可以通过自然语言指令操作安卓手机
# 本项目集成AutoGLM，实现以下功能：
# - 火车票/机票预订（12306、携程等）
# - 网购下单（淘宝、京东、拼多多等）
# - 外卖订餐（美团、饿了么等）
# - 网约车预约（滴滴等）
#
# 工作原理：
# 1. 在服务器上部署虚拟安卓手机（模拟器）
# 2. 用户发起请求后，AutoGLM在虚拟手机上执行APP操作
# 3. 操作完成后，用户在自己真实手机上确认并支付
# 
# 模型服务选择：
# - 智谱BigModel（推荐）: https://open.bigmodel.cn/
# - ModelScope: https://modelscope.cn/models/ZhipuAI/AutoGLM-Phone-9B

# 是否启用 AutoGLM 手机操作功能
AUTOGLM_ENABLED=true

# AutoGLM 模型服务 URL
# 智谱BigModel（推荐）:
AUTOGLM_BASE_URL=https://open.bigmodel.cn/api/paas/v4
# ModelScope（备选）:
# AUTOGLM_BASE_URL=https://api-inference.modelscope.cn/v1

# AutoGLM 模型名称
# 智谱BigModel: autoglm-phone
# ModelScope: ZhipuAI/AutoGLM-Phone-9B
AUTOGLM_MODEL=autoglm-phone

# AutoGLM API Key（从智谱平台或ModelScope平台获取）
AUTOGLM_API_KEY=your_autoglm_api_key_here

# Open-AutoGLM 项目本地路径（可选，用于命令行调用）
# 如果配置了此路径，会通过subprocess调用Open-AutoGLM
# 如果不配置，会使用模拟模式（用于测试）
# AUTOGLM_PATH=/path/to/Open-AutoGLM

# 默认虚拟设备ID（安卓模拟器设备号）
# 可通过 adb devices 命令查看连接的设备
AUTOGLM_DEVICE_ID=emulator-5554

# 任务执行超时时间（秒）
AUTOGLM_TIMEOUT=180

# === 数据库配置 ===
# Redis（可选，用于会话存储）
REDIS_URL=redis://localhost:6379/0

# SQLite（默认）
DATABASE_URL=sqlite:///./ty_mem_agent.db

# === 安全配置 ===
SECRET_KEY=your_super_secret_key_change_in_production
# 访问令牌过期时间（分钟）- 默认43200分钟（30天）
ACCESS_TOKEN_EXPIRE_MINUTES=43200

# === 日志配置（仅保留 LOG_LEVEL 作为唯一入口） ===
LOG_LEVEL=INFO
LOG_FILE=logs/ty_mem_agent.log

# === 记忆配置 ===
MEMORY_MAX_TOKENS=4000
MEMORY_RETENTION_DAYS=30

# === Agent 行为配置 ===
# Agent 主动性级别 (passive/moderate/proactive/aggressive)
# passive: 被动模式 - 只执行明确要求，不主动建议
# moderate: 适中模式 - 适度主动，不会过度打扰
# proactive: 主动模式 - 积极主动服务 (默认)
# aggressive: 激进模式 - 非常主动，提供大量建议
AGENT_PROACTIVITY_LEVEL=proactive

# 启用后，应用启动时会自动将所有APP API接口注册到Nacos服务注册中心

# Nacos服务器地址
# 格式: ip:port 或 ip1:port1,ip2:port2（多个地址用逗号分隔）
NACOS_SERVER_ADDRESSES=localhost:8848

# Nacos命名空间（可选，留空使用public命名空间）
# NACOS_NAMESPACE=your_namespace_id

# Nacos用户名和密码（可选，如果Nacos启用了认证）
# NACOS_USERNAME=nacos
# NACOS_PASSWORD=nacos

# 服务名称（注册到Nacos的服务名）
