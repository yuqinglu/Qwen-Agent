# TY Memory Agent 环境配置示例
# 复制此文件为 .env 并填入您的配置

# === 基础配置 ===
HOST=0.0.0.0
PORT=8080

# === LLM配置 ===
# DashScope配置（推荐）
DASHSCOPE_API_KEY=your_dashscope_api_key_here
DEFAULT_LLM_MODEL=qwen-max

# OpenAI配置（备选）
# OPENAI_API_KEY=your_openai_api_key_here
# OPENAI_BASE_URL=https://api.openai.com/v1

# === ASR 语音识别配置 ===
# 阿里云百炼 Fun-ASR WebSocket API 地址
# 北京地域（默认）
ASR_WS_URL=wss://dashscope.aliyuncs.com/api-ws/v1/inference
# 新加坡地域（如需使用海外服务，取消注释下面这行并注释上面这行）
# ASR_WS_URL=wss://dashscope-intl.aliyuncs.com/api-ws/v1/inference

# ASR 模型名称（默认：fun-asr-realtime）
# 可选值：
# - fun-asr-realtime: 稳定版（当前等同fun-asr-realtime-2025-11-07）
# - fun-asr-realtime-2025-11-07: 快照版，做了远场VAD优化
# - fun-asr-realtime-2025-09-15: 旧版快照
ASR_MODEL=fun-asr-realtime

# 默认音频采样率（Hz）
# 推荐值：16000（Fun-ASR标准采样率）
ASR_SAMPLE_RATE=16000

# 默认音频格式
# 可选值：pcm, wav, mp3, opus, speex, aac, amr
# 推荐值：pcm（用于实时流式识别）
ASR_FORMAT=pcm

# 默认识别语言
# 可选值：zh（中文）, en（英文）, ja（日语）
ASR_LANGUAGE=zh

# === MemOS记忆系统配置 ===
MEMOS_API_BASE=https://api.openmem.net
MEMOS_API_KEY=your_memos_api_key_here

# === MCP服务API密钥 ===
# 滴滴叫车（需要申请）
DIDI_API_KEY=your_didi_api_key_here
DIDI_MCP_MODE=production  # production 或 sandbox

# 高德地图 API Key（必需 - 用于所有高德地图功能）
# 申请地址: https://lbs.amap.com/api/webservice/guide/create-project/get-key
# 功能包括: 天气查询、地理编码、POI搜索、路径规划、距离测量等
AMAP_TOKEN=your_amap_api_key_here

# 博查AI搜索引擎 API Key（可选 - 用于智能搜索功能）
# 申请地址: https://bocha-ai.feishu.cn/wiki/DCZTwH6OMidCbTkWFRVcVkHonag
# 功能包括: Web搜索、AI搜索（需要付费）
BOCHA_API_KEY=your_bocha_api_key_here

# 飞常准航班信息 API Key（可选 - 用于航班信息查询）
# 申请地址: 飞友AI开放平台 https://ai.variflight.com/
# 功能包括: 航班实时动态、行程价格、舒适度、航班行程价格等
VARIFLIGHT_API_KEY=your_variflight_api_key_here

# 饿了么外卖 API 配置（可选 - 用于外卖点餐功能）
# 申请地址: 饿了么开放平台（需联系饿了么获取开发者账号）
# 功能包括: 餐厅搜索、菜单查询、购物车管理、订单创建和管理
# 文档: https://openapi-doc.faas.ele.me/v2/api/index.html
ELEME_APP_KEY=your_eleme_app_key_here
ELEME_APP_SECRET=your_eleme_app_secret_here
ELEME_MODE=sandbox  # sandbox（测试环境） 或 production（生产环境）

# 飞书会议 API 配置（可选 - 用于会议管理功能）
# 申请地址: 飞书开放平台 https://open.feishu.cn/
# 功能包括: 创建会议、更新会议、删除会议、获取会议详情、设置主持人、邀请参会人
# 文档: https://feishu.apifox.cn/api-9020884
# 需要权限: 查看、创建和管理日历和会议
FEISHU_APP_ID=your_feishu_app_id_here
FEISHU_APP_SECRET=your_feishu_app_secret_here

# === 数据库配置 ===
# Redis（可选，用于会话存储）
REDIS_URL=redis://localhost:6379/0

# SQLite（默认）
DATABASE_URL=sqlite:///./ty_mem_agent.db

# === 安全配置 ===
SECRET_KEY=your_super_secret_key_change_in_production
# 访问令牌过期时间（分钟）- 默认43200分钟（30天）
ACCESS_TOKEN_EXPIRE_MINUTES=43200

# === 日志配置（仅保留 LOG_LEVEL 作为唯一入口） ===
LOG_LEVEL=INFO
LOG_FILE=logs/ty_mem_agent.log

# === 记忆配置 ===
MEMORY_MAX_TOKENS=4000
MEMORY_RETENTION_DAYS=30

# === Agent 行为配置 ===
# Agent 主动性级别 (passive/moderate/proactive/aggressive)
# passive: 被动模式 - 只执行明确要求，不主动建议
# moderate: 适中模式 - 适度主动，不会过度打扰
# proactive: 主动模式 - 积极主动服务 (默认)
# aggressive: 激进模式 - 非常主动，提供大量建议
AGENT_PROACTIVITY_LEVEL=proactive

# === Nacos 服务注册配置（可选）===
# 是否启用Nacos服务注册
# 启用后，应用启动时会自动将所有APP API接口注册到Nacos服务注册中心
NACOS_ENABLED=false

# Nacos服务器地址
# 格式: ip:port 或 ip1:port1,ip2:port2（多个地址用逗号分隔）
NACOS_SERVER_ADDRESSES=localhost:8848

# Nacos命名空间（可选，留空使用public命名空间）
# NACOS_NAMESPACE=your_namespace_id

# Nacos用户名和密码（可选，如果Nacos启用了认证）
# NACOS_USERNAME=nacos
# NACOS_PASSWORD=nacos

# 服务名称（注册到Nacos的服务名）
NACOS_SERVICE_NAME=ty-memory-agent

# 服务组名（可选，不设置时Nacos会使用默认值 DEFAULT_GROUP）
# 只有在需要自定义服务分组时才需要设置此参数
# NACOS_GROUP_NAME=DEFAULT_GROUP
